  balance NUMERIC(15, 2) DEFAULT 0.00

CREATE TABLE transfers (
  id SERIAL PRIMARY KEY,
  from_account_id INT REFERENCES accounts(id),
  to_account_id INT REFERENCES accounts(id),
  amount NUMERIC(15, 2) NOT NULL
